## 后端配置文件 application.yml

在 IDEA 的右侧 Maven 面板中，双击 `Lifecycle` -> `clean`，然后双击 `package`。

- **产物**：在 `target` 目录下会生成一个 `.jar` 文件（例如 `lab-monitor-backend-0.0.1-SNAPSHOT.jar`）。
- **操作**：把它重命名为 **app.jar**（为了方便 Dockerfile 统一识别）。

```
server:

```

dockerfile

```dockerfile
# 使用轻量级 JDK 17
FROM openjdk:17-jdk-alpine
# 挂载临时目录
VOLUME /tmp
# 复制 jar 包
COPY app.jar app.jar
# 暴露端口
EXPOSE 8080
# 启动命令
ENTRYPOINT ["java","-jar","/app.jar"]
```



## 前端

**npm run build**打包前端

这会生成一个 `dist` 文件夹。**把 dist 文件夹里的所有内容**，上传到服务器的 `docker/html/dist/` 目录下。

## docker

```dockerfile
version: '3.8'

services:
  # ==========================================
  # 1. MySQL (业务数据库)
  # ==========================================
  mysql:
    image: mysql:8.0
    container_name: lab201_mysql
    restart: always
    # 兼容性设置：使用旧版密码加密方式，防止连接报错
    command: --default-authentication-plugin=mysql_native_password
    environment:
      # 🔒 [安全修改] 请务必修改这里的密码！
      MYSQL_ROOT_PASSWORD: lab201201
      MYSQL_DATABASE: lab201_monitor
      TZ: Asia/Shanghai
    ports:
      - "3306:3306"
    volumes:
      - ./data/mysql:/var/lib/mysql
      - ./sql/init.sql:/docker-entrypoint-initdb.d/init.sql
    networks:
      - lab-net
    healthcheck:
      test: ["CMD", "mysqladmin" ,"ping", "-h", "localhost"]
      interval: 10s
      timeout: 5s
      retries: 5

  # ==========================================
  # 2. Redis (缓存)
  # ==========================================
  redis:
    image: redis:7.0
    container_name: lab201_redis
    restart: always
    # 🔒 [安全修改] 设置强密码
    command: redis-server --requirepass "lab201201" --appendonly yes
    ports:
      - "6379:6379"
    volumes:
      - ./data/redis:/data
    networks:
      - lab-net

  # ==========================================
  # 3. RabbitMQ (消息队列)
  # ==========================================
  rabbitmq:
    image: rabbitmq:3-management
    container_name: lab201_rabbitmq
    restart: always
    ports:
      - "5672:5672"
      - "15672:15672"
    environment:
      # 🔒 [安全修改] 修改默认账号
      RABBITMQ_DEFAULT_USER: lab201admin
      RABBITMQ_DEFAULT_PASS: lab201201
    networks:
      - lab-net

  etcd:
    container_name: milvus-etcd
    image: quay.io/coreos/etcd:v3.5.18
    environment:
      - ETCD_AUTO_COMPACTION_MODE=revision
      - ETCD_AUTO_COMPACTION_RETENTION=1000
      - ETCD_QUOTA_BACKEND_BYTES=4294967296
      - ETCD_SNAPSHOT_COUNT=50000
    volumes:
      - ./data/etcd:/etcd
    command: etcd -advertise-client-urls=http://etcd:2379 -listen-client-urls http://0.0.0.0:2379 --data-dir /etcd
    healthcheck:
      test: ["CMD", "etcdctl", "endpoint", "health"]
      interval: 30s
      timeout: 20s
      retries: 3
    networks:
      - lab-net

    container_name: milvus-minio
    image: minio/minio:RELEASE.2024-05-28T17-19-04Z
    environment:
      MINIO_ACCESS_KEY: minioadmin
      MINIO_SECRET_KEY: minioadmin

      - "9001:9001"
      - "9000:9000"
    volumes:
      - ./data/minio:/minio_data
    command: minio server /minio_data --console-address ":9001"
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:9000/minio/health/live"]
      interval: 30s
      timeout: 20s
      retries: 3
    networks:
      - lab-net

  standalone:
    container_name: milvus-standalone
    image: milvusdb/milvus:v2.5.24
    command: ["milvus", "run", "standalone"]
    security_opt:
      - seccomp:unconfined
    environment:
      MINIO_REGION: us-east-1
      ETCD_ENDPOINTS: etcd:2379
      MINIO_ADDRESS: minio:9000
    volumes:
      - ./data/milvus:/var/lib/milvus
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:9091/healthz"]
      interval: 30s
      start_period: 90s
      timeout: 20s
      retries: 3

      - "19530:19530"
      - "9091:9091"

      - "etcd"
      - "minio"
    networks:
      - lab-net

  # ==========================================
  # 6. Attu (Milvus 可视化管理工具 - 新增)
  # 它可以让你像操作 MinIO 一样在网页上操作 Milvus

      - "8000:3000"

      - standalone

      - lab-net

    container_name: lab201_ollama
    restart: always

      - "11434:11434"

        reservations:
          devices:
            - driver: nvidia
              count: 1
              capabilities: [gpu]
    runtime: nvidia
    environment:
    - NVIDIA_VISIBLE_DEVICES=all

  # # ==========================================
  # # 6. Nginx (门户)
  # # ==========================================
  # nginx:
  #   image: nginx:latest
  #   container_name: lab201_frontend
  #   restart: always
  #   ports:
  #     - "80:80"
  #   volumes:
  #     - ./html:/usr/share/nginx/html
  #     - ./nginx.conf:/etc/nginx/conf.d/default.conf
  #   networks:
  #     - lab-net

  backend:
    build: ./backend
    container_name: lab201_backend
    restart: always
    ports:
      - "8080:8080"
    environment:
      # 连接 docker 里的 mysql 服务
      SPRING_DATASOURCE_URL: jdbc:mysql://mysql:3306/lab201_monitor?useUnicode=true&characterEncoding=utf-8&serverTimezone=Asia/Shanghai&allowPublicKeyRetrieval=true&useSSL=false
      SPRING_DATASOURCE_USERNAME: root
      SPRING_DATASOURCE_PASSWORD: lab201201
      # 连接 docker 里的 redis 服务
      SPRING_REDIS_HOST: redis
      SPRING_REDIS_PASSWORD: lab201201
    depends_on:
      - mysql
      - redis
    networks:
      - lab-net

  # ==========================================
  # [修改] Nginx 前端网关
  # ==========================================
  nginx:
    image: nginx:latest
    container_name: lab201_frontend
    restart: always
    ports:
      - "80:80"
    volumes:
      # 挂载 html 目录 (包含 dist 和 wiki)
      - ./html:/usr/share/nginx/html
      # 挂载配置文件
      - ./nginx.conf:/etc/nginx/nginx.conf
    depends_on:
      - backend
    networks:
      - lab-net

networks:
  lab-net:
    driver: bridge
```



## nginx

```nginx
worker_processes  1;

events {
    worker_connections  1024;
}

http {
    include       mime.types;
    default_type  application/octet-stream;
    sendfile        on;
    keepalive_timeout  65;
    gzip  on;

    server {
        listen       80;
        server_name  localhost;

        # ==============================
        # 1. Vue 前端 (根路径)
        # ==============================
        location / {
            # 对应宿主机 ./html/dist
            root   /usr/share/nginx/html/dist;
            index  index.html index.htm;
            # 解决 Vue 路由刷新 404 问题
            try_files $uri $uri/ /index.html;
        }

        # ==============================
        # 2. MDwiki (子路径)
        # ==============================
        location /wiki/ {
            # 对应宿主机 ./html/wiki
            alias  /usr/share/nginx/html/wiki/;
            index  index.html;
        }

        # ==============================
        # 3. 后端 API 反向代理 (必须加这个！)
        # ==============================
        location /api/v1/ {
            # 'backend' 是 docker-compose 里的服务名
            proxy_pass http://backend:8080/api/v1/;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        }
    }
}
```
