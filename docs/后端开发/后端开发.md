## 接口规范

**路径规范**：所有接口统一以 `/api/v1` 开头（版本控制），资源名使用 **kebab-case**（短横线命名，如 `server-node`），而不是驼峰。

**方法规范**：

- `GET`：查询（单条、列表、分页）
- `POST`：新增 / 复杂业务操作（如上报）
- `PUT`：修改
- `DELETE`：删除

**文档规范**：全面接入 `Knife4j`，每个接口必须有 `@Operation` 说明。

**参数规范**：分页统一用 `page`, `size`，复杂查询用 POST Body 或 Query String。

### 3. 创建 JWT 工具类

根据你提供的代码，创建 `JwtUtil`。

```java
package com.lab201.monitor.common.utils;

import com.lab201.monitor.common.exception.LabException;
import com.lab201.monitor.common.result.ResultCodeEnum;
import io.jsonwebtoken.*;
import io.jsonwebtoken.security.Keys;
import javax.crypto.SecretKey;
import java.util.Date;

public class JwtUtil {
    // 1小时过期，可根据需求调整
    private static long tokenExpiration = 60 * 60 * 1000L; 
    // 签名密钥，长度需符合算法要求
    private static SecretKey tokenSignKey = Keys.hmacShaKeyFor("M0PKKI6pYGVWWfDZw90a0lTpGYX1d4AQ".getBytes());

    // 生成token
    public static String createToken(Long userId, String username,String role) {
        String token = Jwts.builder()
                .setSubject("USER_INFO")
                .setExpiration(new Date(System.currentTimeMillis() + tokenExpiration))
                .setIssuedAt(new Date())
                .claim("userId", userId)
                .claim("username", username)
                .claim("role",role)
                .signWith(tokenSignKey)
                .compact();
        return token;
    }

    // 解析token
    public static Claims parseToken(String token) {
        if (token == null) {
            throw new LabException(ResultCodeEnum.Token_Empty);
        }
        try {
            JwtParser jwtParser = Jwts.parserBuilder().setSigningKey(tokenSignKey).build();
            return jwtParser.parseClaimsJws(token).getBody();
        } catch (ExpiredJwtException e) {
            throw new LabException(ResultCodeEnum.UNAUTHORIZED);
        } catch (JwtException e) {
            throw new LabException(ResultCodeEnum.Token_Error);
        }
    }
}
```

###  4. 实现拦截器

在拦截器中解析 Token 并存入 ThreadLocal。

```java
package com.lab201.monitor.common.config;

import com.lab201.monitor.common.login.LoginUser;
import com.lab201.monitor.common.login.LoginUserHolder;
import com.lab201.monitor.common.utils.JwtUtil;
import io.jsonwebtoken.Claims;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.stereotype.Component;
import org.springframework.web.servlet.HandlerInterceptor;

@Component
public class AuthenticationInterceptor implements HandlerInterceptor {

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) throws Exception {
        // 放行 OPTIONS 请求 (处理跨域预检)
        if ("OPTIONS".equalsIgnoreCase(request.getMethod())) {
            return true;
        }

        // 1. 获取 Token (注意这里用的 header key 是 access-token)
        String token = request.getHeader("access-token");
        
        // 2. 解析 Token
        Claims claims = JwtUtil.parseToken(token);
        Long userId = claims.get("userId", Long.class);
        String username = claims.get("username", String.class);
        String role = claims.get("role",String.class);

        // 3. 存入ThreadLocal
        LoginUserHolder.setLoginUser(new LoginUser(userId, username, role));

        return true;
    }

    @Override
    public void afterCompletion(HttpServletRequest request, HttpServletResponse response, Object handler, Exception ex) throws Exception {
        // 请求结束后清理线程变量，防止内存泄漏
        LoginUserHolder.clear();
    }
}
```

```java
import io.minio.*;

    @Autowired
    private MinioClient minioClient;

    /**
     * 上传文件
     * @return 返回生成的唯一对象名 (UUID.pdf)
     */
    public String upload(String bucketName, MultipartFile file) {

        try {
            if (!minioClient.bucketExists(BucketExistsArgs.builder().bucket(bucketName).build())) {
                minioClient.makeBucket(MakeBucketArgs.builder().bucket(bucketName).build());
                // 可选：设置 Bucket 公共读权限（根据业务需求）
                //minioClient.setBucketPolicy(
                //        SetBucketPolicyArgs.builder()
                //                .bucket(bucketName)
                //                .config("{\"Version\":\"2026-01-01\",\"Statement\":[{\"Effect\":\"Allow\",\"Principal\":\"*\",\"Action\":\"s3:GetObject\",\"Resource\":\"arn:aws:s3:::" + bucketName + "/*\"}]}")
                //                .build()
                //);
            }
        } catch (Exception e) {
            throw new RuntimeException(e);
        }

        String originalFilename = file.getOriginalFilename();
        if (originalFilename == null) originalFilename = "unknown.file";
        
        // 生成唯一文件名，防止重名覆盖
        String suffix = originalFilename.contains(".") ? 
                        originalFilename.substring(originalFilename.lastIndexOf(".")) : "";
        String objectName = UUID.randomUUID().toString().replace("-", "") + suffix;

        try {
            InputStream inputStream = file.getInputStream();
            minioClient.putObject(
                    PutObjectArgs.builder()
                            .bucket(bucketName)
                            .object(objectName)
                            .stream(inputStream, file.getSize(), -1)
                            .contentType(file.getContentType())
                            .build()
            );
            return objectName;
        } catch (Exception e) {
            log.error("MinIO上传失败", e);
            throw new RuntimeException("文件上传MinIO失败");
        }
    }

    /**
     * 获取文件预览/下载链接 (签名URL，有效期默认1小时)
     * 这样前端拿到的是一个 http://192.168.31.130:9000/... 的临时链接
     */
    public String getPresignedUrl(String bucketName, String objectName) {
        try {
            return minioClient.getPresignedObjectUrl(
                    GetPresignedObjectUrlArgs.builder()
                            .method(Method.GET)
                            .bucket(bucketName)
                            .object(objectName)
                            .expiry(1, TimeUnit.HOURS) // 1小时有效
                            .build()
            );
        } catch (Exception e) {
            log.error("获取MinIO链接失败", e);
            return null;
        }
    }

    /**
     * 删除文件
     */
    public void remove(String bucketName, String objectName) {
        try {
            minioClient.removeObject(
                    RemoveObjectArgs.builder().bucket(bucketName).object(objectName).build()
            );
            log.info("文件已删除: bucket={}, fileName={}", bucketName, objectName);
        } catch (Exception e) {
            log.error("MinIO删除失败", e);
        }
    }

    public InputStream getObject(String bucketName, String objectName) {
        try {
            return minioClient.getObject(
                    GetObjectArgs.builder()
                            .bucket(bucketName)
                            .object(objectName)
                            .build()
            );
        } catch (Exception e) {

        }
    }
```

```java
package com.lab201.monitor.service;

import com.lab201.monitor.common.event.RagParseEvent;
import com.lab201.monitor.common.utils.MinioUtil;
import com.lab201.monitor.entity.SysDocument;
import lombok.extern.slf4j.Slf4j;
import org.springframework.ai.chat.client.ChatClient;
import org.springframework.ai.document.Document;
import org.springframework.ai.reader.tika.TikaDocumentReader;
import org.springframework.ai.transformer.splitter.TokenTextSplitter;
import org.springframework.ai.vectorstore.SearchRequest;
import org.springframework.ai.vectorstore.VectorStore;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.context.event.EventListener;
import org.springframework.core.io.InputStreamResource;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Service;
import org.springframework.transaction.event.TransactionPhase;
import org.springframework.transaction.event.TransactionalEventListener;

import java.io.InputStream;
import java.util.List;
import java.util.stream.Collectors;

@Slf4j
@Service
public class RagService {

    @Autowired
    private ISysDocumentService documentService;

    @Autowired
    private MinioUtil minioUtil;

    @Autowired
    private VectorStore vectorStore;

    @Autowired
    private ChatClient chatClient;

    /**
     * 1. 异步监听文档解析事件 (入库流程)
     */
    @Async
    @TransactionalEventListener(phase = TransactionPhase.AFTER_COMMIT)
    public void handleRagParse(RagParseEvent event) {
        Long docId = event.getDocumentId();
        log.info(">>> [RAG] 开始解析文档 ID: {}", docId);

        SysDocument doc = documentService.getById(docId);
        if (doc == null) return;

        try {
            // 1.1 更新状态：解析中
            doc.setParseStatus(1);
            documentService.updateById(doc);

            // 1.2 从 MinIO 获取文件流
            InputStream inputStream = minioUtil.getObject(doc.getBucketName(), doc.getObjectName());

            // 1.3 【Extract】读取并解析文档
            TikaDocumentReader reader = new TikaDocumentReader(new InputStreamResource(inputStream));
            List<Document> documents = reader.get();

            // 1.4 【Transform】文本切片
            TokenTextSplitter splitter = new TokenTextSplitter();
            List<Document> splitDocuments = splitter.apply(documents);

            // 1.5 【Metadata】补充元数据
            for (Document splitDoc : splitDocuments) {
                splitDoc.getMetadata().put("doc_id", docId);
                splitDoc.getMetadata().put("title", doc.getTitle());
                splitDoc.getMetadata().put("view_scope", doc.getViewScope());
            }

            // 1.6 【Load】一键存入
            vectorStore.add(splitDocuments);

            // 1.7 更新状态：成功
            doc.setParseStatus(2);
            // 🔥🔥🔥 修正点 1：使用 getText() 获取内容 🔥🔥🔥
            if (!splitDocuments.isEmpty()) {
                String content = splitDocuments.get(0).getText(); // 改为 getText()
                // 增加空指针保护，防止内容为空
                if (content != null) {
                    doc.setSummary(content.substring(0, Math.min(content.length(), 200)));
                } else {
                    doc.setSummary("无文本内容");
                }
            }
            documentService.updateById(doc);

            log.info(">>> [RAG] 文档解析完成 ID: {}, 切片数: {}", docId, splitDocuments.size());

        } catch (Exception e) {
            log.error(">>> [RAG] 文档解析失败 ID: {}", docId, e);
            doc.setParseStatus(3);
            documentService.updateById(doc);
        }
    }

    /**
     * 2. 对话流程 (检索 + 生成)
     */
    public String chat(String query) {
        // 2.1 【Retrieve】检索
        List<Document> similarDocuments = vectorStore.similaritySearch(
                SearchRequest.builder().query(query).topK(5).build()
        );

        // 2.2 【Context】组装上下文
        // 🔥🔥🔥 修正点 2：使用 Document::getText 🔥🔥🔥
        String context = similarDocuments.stream()
                .map(Document::getText) // 改为 getText
                .collect(Collectors.joining("\n\n"));

        // 2.3 【Generate】生成回答
        String prompt = """
                基于以下已知信息回答用户的问题。如果无法从信息中得到答案，请说“不知道”。
                
                --- 已知信息 ---
                {context}
                --- 信息结束 ---
                
                问题：{query}
                """;

        return chatClient.prompt()
                .user(u -> u.text(prompt)
                        .param("context", context)
                        .param("query", query))
                .call()
                .content();
    }
}
```

1. 原因深度解析

现在的逻辑是**“严格 RAG 模式”**。

1. **在 AiConfig 中**：你告诉它系统人设是 `...如果上下文没有相关信息，请诚实地回答不知道，不要编造。` —— 这是一道**死命令**。
2. **在 RagService 中**：你又强调了一遍 `...如果无法从信息中得到答案，请说“不知道”。` —— 这是第二道**死命令**。

当用户问“介绍西安电子科技大学”时：

1. 系统去 Milvus 检索，可能只找到了“李玉钊是西安电子科技大学的学生”这句话。
2. AI 发现检索到的内容里没有“大学的详细介绍”。
3. AI 想要调用自己的通用知识库（由于它本来就知道西电），但它想起了你的两条“死命令”。
4. 于是它严格遵守指令，回答：“不知道”。

2. 解决方案：改为“混合增强模式”

需要修改提示词，告诉 AI：**“优先参考给你的资料，如果资料里没有，就用你自己的知识回答。”**

需要同时修改 `AiConfig` 和 `RagService`。

```java
     * 2. 对话流程 (检索 + 生成)

    public String chat(String query) {
        // 2.1 【Retrieve】检索
        List<Document> similarDocuments = vectorStore.similaritySearch(
                SearchRequest.builder().query(query).topK(5).build()
        );

        // 2.2 【Context】组装上下文
        // 🔥🔥🔥 修正点 2：使用 Document::getText 🔥🔥🔥
        String context = similarDocuments.stream()
                .map(Document::getText) // 改为 getText
                .collect(Collectors.joining("\n\n"));

        // 2.3 【Generate】生成回答
        //String prompt = """
        //        基于以下已知信息回答用户的问题。如果无法从信息中得到答案，请说“不知道”。
        //
        //        --- 已知信息 ---
        //        {context}
        //        --- 信息结束 ---
        //
        //        问题：{query}
        //        """;
        String prompt = """
            请回答用户的问题。
            
            参考信息如下（如果有）：
            --- 上下文开始 ---
            {context}
            --- 上下文结束 ---
            
            回答要求：
            1. 如果参考信息中有答案，请直接基于参考信息回答。
            2. 如果参考信息中没有答案，请使用你自己的通用知识来回答。
            3. 请保持语气亲切、专业。
            
            用户问题：{query}
            """;

        return chatClient.prompt()
                .user(u -> u.text(prompt)
                        .param("context", context)
                        .param("query", query))
                .call()
                .content();
    }

    public Flux<String> chatStream(String query) {
        // 1. 【检索】(跟之前一样，先查资料)
        List<Document> similarDocuments = vectorStore.similaritySearch(
                SearchRequest.builder().query(query).topK(5).build()
        );

        String context = similarDocuments.stream()
                .map(Document::getText)
                .collect(Collectors.joining("\n\n"));

        // 2. 【生成】(使用 stream() 而不是 call())
        String prompt = """
            请回答用户的问题。
            
            参考信息如下（如果有）：
            --- 上下文开始 ---
            {context}
            --- 上下文结束 ---
            
            回答要求：
            1. 如果参考信息中有答案，请直接基于参考信息回答。
            2. 如果参考信息中没有答案，请使用你自己的通用知识来回答。
            3. 请保持语气亲切、专业。
            
            用户问题：{query}
            """;

        // 🔥 核心区别：使用 .stream().content() 返回 Flux<String>
        return chatClient.prompt()
                .user(u -> u.text(prompt)
                        .param("context", context)
                        .param("query", query))
                .stream()
                .content();
    }
```
