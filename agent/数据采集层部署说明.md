### 第一部分：运行方式选择

> **问题**：如果实现定时上报是一直运行 py 脚本吗？后台实现吗？还是 crontab 定时任务？

**结论：强烈建议使用 Systemd 服务（守护进程）一直运行 while True 脚本。**

- **❌ Crontab (不推荐)**：
  - Crontab 最小粒度是 **1分钟**。对于 GPU 监控，1分钟太慢了（显存可能几秒钟就飙升又下降）。
  - 每次启动 Python 都要重新初始化 NVML，开销大。
- **✅ Systemd / 常驻进程 (推荐)**：
  - **实时性高**：你可以设置 `time.sleep(3)`，每 3 秒上报一次，非常丝滑。
  - **稳定性**：使用 Linux 的 `Systemd` 管理，脚本挂了会自动重启，开机自动启动。

```
import pynvml
import psutil
import time

def get_process_user(pid):
    """根据PID获取用户名"""
    try:
        proc = psutil.Process(pid)
        return proc.username()
    except (psutil.NoSuchProcess, psutil.AccessDenied):
        return None

def test_collection():
    try:
        pynvml.nvmlInit()
        device_count = pynvml.nvmlDeviceGetCount()
        print(f"✅ 检测到 {device_count} 张显卡\n")
        print(f"{'索引':<5} {'显卡型号':<20} {'显存(MB)':<15} {'利用率%':<10} {'用户'}")
        print("-" * 70)

        for i in range(device_count):
            handle = pynvml.nvmlDeviceGetHandleByIndex(i)
            name = pynvml.nvmlDeviceGetName(handle)
            # 兼容不同版本的pynvml返回bytes还是str
            if isinstance(name, bytes):
                name = name.decode('utf-8')
                
            mem_info = pynvml.nvmlDeviceGetMemoryInfo(handle)
            util = pynvml.nvmlDeviceGetUtilizationRates(handle).gpu
            
            # 获取占用显存最大的用户
            procs = pynvml.nvmlDeviceGetComputeRunningProcesses(handle)
            current_user = "Idle"
            
            if len(procs) > 0:
                # 找到显存占用最大的进程
                max_proc = max(procs, key=lambda p: p.usedGpuMemory)
                user = get_process_user(max_proc.pid)
                if user:
                    current_user = user

            print(f"{i:<5} {name:<20} {int(mem_info.used/1024/1024):<15} {util:<10} {current_user}")

        pynvml.nvmlShutdown()
        print("\n✅ 测试成功！可以进行下一步。")

    except Exception as e:
        print(f"❌ 测试失败: {e}")
        import traceback
        traceback.print_exc()

if __name__ == "__main__":
    test_collection()
```

### 第三部分：正式上报代码 (适配 Java Controller)

直接用 HTTP POST 发送数据，简单、直接、有效。

```python
import pynvml
import requests
import time
import os
import pwd
import socket
import logging

# ================= 配置区域 =================
# 1. 后端服务器地址 (你的ServerH1 IP)
MASTER_IP = "192.168.31.129" 
# 2. 接口地址 (必须与 Java Controller 的 @RequestMapping 一致)
REPORT_URL = f"http://{MASTER_IP}:8080/api/gpu/report"

# 3. 本机名字 (必须与数据库 server_node 表的 server_name 一致)
# 如果 Linux hostname 就是 'titanxp'，则不用改；否则请手动写死，例如: SERVER_NAME = "titanxp"
SERVER_NAME = socket.gethostname() 
# ===========================================

# 配置日志打印
logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')

def get_gpu_user(handle):
    """获取显存占用最高的用户名"""
    try:
        procs = pynvml.nvmlDeviceGetComputeRunningProcesses(handle)
        if not procs: return "" # 对应Java的空字符串
        
        # 找到显存占用最大的进程
        max_proc = max(procs, key=lambda p: p.usedGpuMemory)
        
        # 读取 /proc/{pid} 获取用户信息
        try:
            uid = os.stat(f"/proc/{max_proc.pid}").st_uid
            return pwd.getpwuid(uid).pw_name
        except (FileNotFoundError, ProcessLookupError):
            return "" # 进程可能刚结束
    except Exception as e:
        logging.error(f"获取用户失败: {e}")
        return ""

def collect_and_report():
    try:
        pynvml.nvmlInit()
        device_count = pynvml.nvmlDeviceGetCount()
        logs = [] # 对应 Java 的 List<GpuMetricLog>

        for i in range(device_count):
            handle = pynvml.nvmlDeviceGetHandleByIndex(i)
            mem_info = pynvml.nvmlDeviceGetMemoryInfo(handle)
            util_rates = pynvml.nvmlDeviceGetUtilizationRates(handle)
            
            # 获取显卡名 (处理 bytes/str 兼容性)
            name_raw = pynvml.nvmlDeviceGetName(handle)
            gpu_name = name_raw.decode('utf-8') if isinstance(name_raw, bytes) else name_raw

            # 构造单条显卡数据 (字段名必须与 Java GpuMetricLog 实体类完全一致!)
            logs.append({
                "gpuIndex": i,
                "gpuName": gpu_name,
                "utilization": util_rates.gpu,
                "memoryUsed": int(mem_info.used / 1024 / 1024),
                "memoryTotal": int(mem_info.total / 1024 / 1024),
                "temperature": pynvml.nvmlDeviceGetTemperature(handle, pynvml.NVML_TEMPERATURE_GPU),
                "fanSpeed": pynvml.nvmlDeviceGetFanSpeed(handle),
                "powerUsage": pynvml.nvmlDeviceGetPowerUsage(handle) // 1000,
                "userName": get_gpu_user(handle)
            })

        pynvml.nvmlShutdown()

        # 组装发送给 Java 的包 (对应 Java 的 GpuReportDto)
        payload = {
            "serverName": SERVER_NAME,
            "logs": logs
        }

        # 发送 POST 请求
        logging.info(f"正在上报 [{SERVER_NAME}] 的数据到 {REPORT_URL}...")
        resp = requests.post(REPORT_URL, json=payload, timeout=3)
        
        if resp.status_code == 200:
            logging.info(f"✅ 上报成功! ({len(logs)} 张显卡)")
        else:
            logging.error(f"❌ 上报失败: 状态码 {resp.status_code} - 返回: {resp.text}")

    except requests.exceptions.ConnectionError:
        logging.error(f"❌ 无法连接到服务器 {MASTER_IP}，请检查防火墙或网络。")
    except Exception as e:
        logging.error(f"❌ 发生未知错误: {e}")

if __name__ == "__main__":
    logging.info(f"🚀 监控 Agent 已启动 | 目标服务器: {MASTER_IP}")
    while True:
        collect_and_report()
        time.sleep(5) # 每5秒上报一次
```

### 第四部分：如何在服务器上部署 (Systemd)

为了让这个脚本开机自启、挂了自动重启，请按照以下步骤操作：

1. **保存脚本**： 假设脚本路径为 `/opt/monitor/gpu_agent.py`。

2. **创建 Systemd 服务文件**： `sudo vim /etc/systemd/system/gpu-monitor.service`

   写入以下内容（注意修改 `User` 和脚本路径）：

```bash
[Unit]
Description=Lab201 GPU Monitor Agent
After=network.target

[Service]
# 修改为你的 Linux 用户名
User=lab201
# 你的 python 路径 和 脚本路径
ExecStart=/home/public_conda/bin/python3 /home/lab201/lab-monitor-platform/gpu-monitor-agent/gpu_monitor.py
# 挂了自动重启
Restart=always
RestartSec=5
# 输出日志到 syslog
StandardOutput=syslog
StandardError=syslog

[Install]
WantedBy=multi-user.target
```

**3.启动服务**：

```bash
sudo systemctl daemon-reload
sudo systemctl enable gpu-monitor
sudo systemctl start gpu-monitor
```

**4.查看日志与服务状态：**

```bash
#查看日志
sudo journalctl -u gpu-monitor -f
#查看状态
sudo systemctl status gpu-monitor
```

**5.重启服务：**
（1）如果了修改python脚本，记得重启服务

```bash
sudo systemctl restart gpu-monitor
```

（2）如果修改了服务配置文件，记得**重载配置**（告诉 Systemd 你改了配置），需要两步

```bash
# 重载配置
sudo systemctl daemon-reload
# 重启服务
sudo systemctl restart gpu-monitor
```
